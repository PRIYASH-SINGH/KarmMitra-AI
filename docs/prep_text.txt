3. Document Ingestion & Chunking (app/ingestion.py) 
This script reads official MoSPI statistical manuals (e.g., CAPI Operations, Survey Sampling 
Methods), cleans the text, applies a sliding character overlap to preserve mathematical and 
operational context, and embeds the chunks into persistent ChromaDB storage using 
all-MiniLM-L6-v2. 
import os 
from langchain_community.document_loaders import PyPDFDirectoryLoader 
from langchain_text_splitters import RecursiveCharacterTextSplitter 
from langchain_community.embeddings import HuggingFaceEmbeddings 
from langchain_community.vectorstores import Chroma 
 
RAW_PDF_DIR = "./data/raw_pdfs" 
PERSIST_DIR = "./vectorstore" 
 
def run_ingestion(): 
    if not os.path.exists(RAW_PDF_DIR): 
        os.makedirs(RAW_PDF_DIR) 
        print(f"Created directory {RAW_PDF_DIR}. Place MoSPI/NSSTA PDFs here.") 
        return 
 
    print("Step 1: Loading PDF documents from disk...") 
    loader = PyPDFDirectoryLoader(RAW_PDF_DIR) 
    raw_docs = loader.load() 
    print(f"Extracted {len(raw_docs)} pages from source manuals.") 
 
    print("Step 2: Performing recursive context-aware chunking...") 
    text_splitter = RecursiveCharacterTextSplitter( 
        chunk_size=750, 
        chunk_overlap=150, 
        length_function=len, 
        separators=["\n\n", "\n", " ", ""] 
    ) 
    chunks = text_splitter.split_documents(raw_docs) 
    print(f"Constructed {len(chunks)} contextual text chunks.") 
 
    print("Step 3: Generating local vector embeddings (all-MiniLM-L6-v2)...") 
    embeddings = HuggingFaceEmbeddings( 
        model_name="sentence-transformers/all-MiniLM-L6-v2", 
        model_kwargs={"device": "cuda"} # Fallback to "cpu" if no GPU available 
    ) 
 
 

 
    print("Step 4: Persisting embeddings into ChromaDB...") 
    vector_db = Chroma.from_documents( 
        documents=chunks, 
        embedding=embeddings, 
        persist_directory=PERSIST_DIR 
    ) 
    print("Vector database successfully initialized and persisted.") 
 
if __name__ == "__main__": 
    run_ingestion() 
 
4. Grounded Assessment Schemas & Generator (app/generator.py) 
To prevent hallucinations, the output format is locked using Pydantic validation. The system 
prompt explicitly forbids using external memory, forcing the model to generate questions directly 
and solely from the retrieved MoSPI context chunks. 
from pydantic import BaseModel, Field 
from typing import List 
import json 
import httpx 
 
# 1. Strict Pydantic Data Contract 
class MCQOption(BaseModel): 
    key: str = Field(description="Option letter: A, B, C, or D") 
    text: str = Field(description="The answer text for this option") 
 
class GeneratedQuestion(BaseModel): 
    competency_code: str = Field(description="The KCM competency code being evaluated") 
    question_text: str = Field(description="The question based solely on the provided text") 
    options: List[MCQOption] = Field(description="Exactly 4 options: A, B, C, and D") 
    correct_option: str = Field(description="Key of the correct option (e.g. 'A')") 
    justification: str = Field(description="Direct citation and explanation from the source text") 
 
class AssessmentPayload(BaseModel): 
    questions: List[GeneratedQuestion] 
 
# 2. Sovereign Local Inference Driver 
class SovereignAIEngine: 
    def __init__(self, host: str = "http://localhost:11434", model: str = "llama3:8b"): 
        # Points to local vLLM or Ollama instance (air-gapped) 
        self.endpoint = f"{host}/api/generate" 
        self.model = model 
 
    def build_prompt(self, context: str, competency_name: str, competency_code: str, count: int = 
3) -> str: 
        return f""" 
You are an expert psychometric evaluator and statistical trainer for the National Statistical 
Systems Training Academy (NSSTA), MoSPI. 
 
 

 
YOUR TASK: 
Generate exactly {count} multiple-choice questions assessing the competency: 
'{competency_name}' (Code: {competency_code}). 
 
STRICT FACTUAL CONSTRAINTS (ZERO-HALLUCINATION POLICY): 
1. Rely SOLELY on the provided Source Context below. 
2. Do NOT extrapolate, assume, or bring in outside statistical training. 
3. Every option must be verifiably grounded in the text. 
4. Output MUST BE strictly valid JSON adhering to this JSON schema: 
{{ 
  "questions": [ 
    {{ 
      "competency_code": "{competency_code}", 
      "question_text": "...", 
      "options": [ 
        {{"key": "A", "text": "..."}}, 
        {{"key": "B", "text": "..."}}, 
        {{"key": "C", "text": "..."}}, 
        {{"key": "D", "text": "..."}} 
      ], 
      "correct_option": "A", 
      "justification": "Exact sentence or rule quoted from the text." 
    }} 
  ] 
}} 
 
SOURCE CONTEXT: 
\"\"\" 
{context} 
\"\"\" 
 
Return ONLY the raw JSON object. Do not include introductory text, explanations, or markdown 
fences. 
""" 
 
    async def generate_mcqs(self, context: str, competency_name: str, competency_code: str, 
count: int = 3) -> AssessmentPayload: 
        prompt = self.build_prompt(context, competency_name, competency_code, count) 
 
        async with httpx.AsyncClient(timeout=60.0) as client: 
            response = await client.post( 
                self.endpoint, 
                json={ 
                    "model": self.model, 
                    "prompt": prompt, 
                    "stream": False, 
                    "format": "json" 
                } 
 

 
            )

@lti_router.get("/login") 
async def oidc_login_initiation( 
    iss: str, 
    login_hint: str, 
    target_link_uri: str 
): 
    """ 
    Step 1: iGOT initiates login. 
    Step 2: KarmMitra validates platform issuer and returns an OpenID Connect redirect. 
    """ 
    if iss != settings.PLATFORM_ISSUER: 
        raise HTTPException(status_code=400, detail="Untrusted platform issuer") 
 
    # Build authentication redirect back to platform authorization endpoint 
    auth_redirect_url = ( 
        f"{settings.PLATFORM_AUTH_URL}?" 
        f"response_type=id_token&" 
        f"scope=openid&" 
        f"client_id={settings.PLATFORM_CLIENT_ID}&" 
        f"redirect_uri={target_link_uri}&" 
        f"login_hint={login_hint}&" 
        f"nonce=sih2026securetokennonce123&" 
        f"prompt=none&" 
        f"response_mode=form_post" 
    ) 
    return RedirectResponse(url=auth_redirect_url) 
 
 
 

 
@lti_router.post("/launch") 
async def lti_message_launch( 
    request: Request, 
    id_token: str = Form(...), 
    state: str = Form(None) 
): 
    """ 
    Step 3: Platform POSTs signed JWT payload containing user role & custom FRAC mapping. 
    """ 
    try: 
        # In mock/dev, unverified decoding allows structural testing 
        # In production, verify against settings.PLATFORM_KEYSET_URL 
        decoded_claims = jwt.decode( 
            id_token,  
            options={"verify_signature": False},  
            algorithms=["RS256"] 
        ) 
 
        # Extract Official's Identity and Custom MoSPI / FRAC Claims 
        user_id = decoded_claims.get("sub") 
        custom_claims = decoded_claims.get("https://purl.imsglobal.org/spec/lti/claim/custom", {}) 
         
        # Pull the FRAC role assigned by MoSPI/iGOT 
        frac_role_code = custom_claims.get("frac_role", "MOSPI_FOD_INV_01") 
        ags_claim = decoded_claims.get("https://purl.imsglobal.org/spec/lti-ags/claim/endpoint", {}) 
        lineitem_url = ags_claim.get("lineitem") 
 
        # Redirect the official directly into the React Learner Portal with a session query 
        frontend_redirect_url = ( 
            f"http://localhost:3000/learner?" 
            f"user_id={user_id}&" 
            f"role={frac_role_code}&" 
            f"lineitem={lineitem_url}" 
        ) 
        return RedirectResponse(url=frontend_redirect_url, status_code=303) 
 
    except Exception as e: 
        raise HTTPException(status_code=401, detail=f"LTI launch validation failed: {str(e)}")

5. Assignment & Grade Services (AGS) Passback (app/lti/ags.py) 
This service implements the LTI 1.3 Score Publish API. When the official finishes an 
assessment, this module sends an encrypted grade payload back to the platform without

import time 
import httpx 
import jwt 
from datetime import datetime, timezone 
from app.core.config import settings 
 
class LTIAdvantageAGSClient: 
    def __init__(self): 
        with open("certs/private.key", "r") as f: 
            self.private_key = f.read() 
 
    def generate_tool_jwt(self) -> str: 
        """Signs an M2M OAuth2 client credentials JWT to obtain access token from iGOT.""" 
        payload = { 
            "iss": settings.PLATFORM_CLIENT_ID, 
            "sub": settings.PLATFORM_CLIENT_ID, 
            "aud": settings.PLATFORM_TOKEN_URL, 
            "iat": int(time.time()), 
            "exp": int(time.time()) + 300, 
            "jti": f"jwt-{int(time.time())}" 
        } 
        return jwt.encode(payload, self.private_key, algorithm="RS256") 
 
    async def submit_competency_score( 
        self,  
        lineitem_url: str,  
        user_id: str,  
        score_achieved: float,  
        score_max: float = 100.0 
    ) -> bool: 
        """ 
        Sends the verified assessment score directly to the iGOT gradebook lineitem. 
        """ 
        if not lineitem_url: 
            return False 
 
        # In production, exchange self.generate_tool_jwt() for bearer token at 
PLATFORM_TOKEN_URL 
        headers = { 
            "Content-Type": "application/vnd.ims.lis.v1.score+json", 
            "Authorization": f"Bearer mock_platform_auth_token" 
        } 
 
 

 
        score_payload = { 
            "timestamp": datetime.now(timezone.utc).isoformat(), 
            "scoreGiven": score_achieved, 
            "scoreMaximum": score_max, 
            "comment": "Completed KarmMitra Diagnostic Assessment", 
            "activityProgress": "Completed", 
            "gradingProgress": "FullyGraded", 
            "userId": user_id 
        } 
 
        score_endpoint = f"{lineitem_url}/scores" 
        async with httpx.AsyncClient(timeout=10.0) as client: 
            try: 
                response = await client.post(score_endpoint, json=score_payload, headers=headers) 
                return response.status_code in [200, 201, 204] 
            except httpx.RequestError: 
                return False 
 
6. Bhashini Translation DPI Integration (app/services/bhashini.py) 
This service translates quiz prompts, UI labels, and feedback on the fly into Indian official 
languages (Hindi, Tamil, Bengali, Marathi, etc.) using the Bhashini NMT API pipeline. 
import httpx 
from typing import Dict, Any 
 
class BhashiniTranslator: 
    def __init__(self, user_id: str = "mock_user", api_key: str = "mock_key"): 
        self.user_id = user_id 
        self.api_key = api_key 
        self.endpoint = "https://dhruva-api.bhashini.gov.in/services/inference/pipeline" 
 
    async def translate_text(self, text: str, source_lang: str = "en", target_lang: str = "hi") -> str: 
        """ 
        Translates text from English to regional languages using Bhashini's official schema. 
        Falls back to original text if connection is unavailable. 
        """ 
        if target_lang == "en" or not text.strip(): 
            return text 
 
        payload = { 
            "pipelineTasks": [ 
                { 
                    "taskType": "translation", 
                    "config": { 
                        "language": { 
                            "sourceLanguage": source_lang, 
                            "targetLanguage": target_lang 
                        } 
                    } 
 

 
                } 
            ], 
            "inputData": { 
                "input": [{"source": text}] 
            } 
        } 
 
        headers = { 
            "Content-Type": "application/json", 
            "userID": self.user_id, 
            "ulcaApiKey": self.api_key 
        } 
 
        async with httpx.AsyncClient(timeout=5.0) as client: 
            try: 
                response = await client.post(self.endpoint, json=payload, headers=headers) 
                if response.status_code == 200: 
                    data = response.json() 
                    return data["pipelineResponse"][0]["output"][0]["target"] 
            except Exception: 
                # Fallback to source string gracefully in offline/hackathon scenarios 
                pass 
        return text

5. Baseline Diagnostic Triage Engine (src/components/BaselineTriage.jsx) 
 

 
This component renders the 5 rapid questions to solve the "cold start" problem. It calculates the 
initial KCM gap scores and invokes the callback to transition to the personalized dashboard 
view. 
import React, { useState } from 'react'; 
import {  
  Box, Card, CardContent, Typography, Button, Radio,  
  RadioGroup, FormControlLabel, FormControl, LinearProgress  
} from '@mui/material'; 
 
export default function BaselineTriage({ questions, onComplete }) { 
  const [currentIndex, setCurrentIndex] = useState(0); 
  const [selectedAnswers, setSelectedAnswers] = useState({}); 
 
  const currentQ = questions[currentIndex]; 
  const progressPercent = ((currentIndex + 1) / questions.length) * 100; 
 
  const handleSelect = (e) => { 
    setSelectedAnswers({ 
      ...selectedAnswers, 
      [currentQ.id]: e.target.value, 
    }); 
  }; 
 
  const handleNext = () => { 
    if (currentIndex + 1 < questions.length) { 
      setCurrentIndex(currentIndex + 1); 
    } else { 
      // Calculate basic proficiency score based on correct options 
      let score = 0; 
      questions.forEach((q) => { 
        if (selectedAnswers[q.id] === q.correctOption) { 
          score += 20; // 5 questions * 20 = 100 points 
        } 
      }); 
      onComplete(score, selectedAnswers); 
    } 
  }; 
 
  return ( 
    <Card elevation={2} sx={{ maxWidth: 700, margin: '2rem auto', p: 2 }}> 
      <CardContent> 
        <Box sx={{ mb: 2 }}> 
          <Typography variant="overline" color="text.secondary"> 
            Baseline Diagnostic Triage • Question {currentIndex + 1} of {questions.length} 
          </Typography> 
          <LinearProgress variant="determinate" value={progressPercent} sx={{ height: 8, 
 

 
borderRadius: 4, mt: 1 }} /> 
        </Box> 
 
        <Typography variant="h6" sx={{ mb: 3 }}> 
          {currentQ.questionText} 
        </Typography> 
 
        <FormControl component="fieldset" sx={{ width: '100%' }}> 
          <RadioGroup value={selectedAnswers[currentQ.id] || ''} onChange={handleSelect}> 
            {currentQ.options.map((opt) => ( 
              <Box  
                key={opt.key}  
                sx={{  
                  border: '1px solid #e2e8f0',  
                  borderRadius: 2,  
                  p: 1,  
                  mb: 1.5, 
                  '&:hover': { bgcolor: '#f8fafc' } 
                }} 
              > 
                <FormControlLabel 
                  value={opt.key} 
                  control={<Radio color="primary" />} 
                  label={`${opt.key}. ${opt.text}`} 
                  sx={{ width: '100%', ml: 0 }} 
                /> 
              </Box> 
            ))} 
          </RadioGroup> 
        </FormControl> 
 
        <Box sx={{ display: 'flex', justifyContent: 'flex-end', mt: 3 }}> 
          <Button  
            variant="contained"  
            color="primary"  
            size="large" 
            disabled={!selectedAnswers[currentQ.id]} 
            onClick={handleNext} 
          > 
            {currentIndex + 1 === questions.length ? 'Finalize Triage' : 'Next Question'} 
          </Button> 
        </Box> 
      </CardContent> 
    </Card> 
  ); 
}

5. Mock iGOT Server Implementation (server.js) 
This script handles the platform-side authorization flow, serves the public keys, signs JWTs, and 
receives score passbacks from KarmMitra: 
const express = require('express'); 
const cors = require('cors'); 
const bodyParser = require('body-parser'); 
const jwt = require('jsonwebtoken'); 
const fs = require('fs'); 
const path = require('path'); 
 
const app = express(); 
 

 
const PORT = 9000; 
 
app.use(cors()); 
app.use(bodyParser.urlencoded({ extended: true })); 
app.use(bodyParser.json()); 
app.use(express.static('public')); 
 
const privateKey = fs.readFileSync(path.join(__dirname, 'certs/platform_private.key'), 'utf8'); 
const publicKey = fs.readFileSync(path.join(__dirname, 'certs/platform_public.key'), 'utf8'); 
const users = JSON.parse(fs.readFileSync(path.join(__dirname, 'data/synthetic_users.json'), 
'utf8')); 
 
// 1. JWKS Endpoint (Used by Member 3 to verify launch signatures) 
app.get('/.well-known/jwks.json', (req, res) => { 
  res.json({ 
    keys: [ 
      { 
        kty: 'RSA', 
        alg: 'RS256', 
        use: 'sig', 
        kid: 'mock-platform-key-1', 
        key: publicKey 
      } 
    ] 
  }); 
}); 
 
// 2. Fetch seeded official personas for the demo portal UI 
app.get('/api/users', (req, res) => { 
  res.json(users); 
}); 
 
// 3. OIDC Launch Simulator: Generates signed LTI 1.3 JWT 
app.post('/platform/launch', (req, res) => { 
  const { userId } = req.body; 
  const user = users.find(u => u.sub === userId) || users[0]; 
 
  const payload = { 
    iss: 'https://igotkarmayogi.gov.in', 
    sub: user.sub, 
    aud: 'karmmitra_sih26101_tool', 
    exp: Math.floor(Date.now() / 1000) + 3600, 
    iat: Math.floor(Date.now() / 1000), 
    name: user.name, 
    email: user.email, 
    'https://purl.imsglobal.org/spec/lti/claim/message_type': 'LtiResourceLinkRequest', 
    'https://purl.imsglobal.org/spec/lti/claim/version': '1.3.0', 
    'https://purl.imsglobal.org/spec/lti/claim/custom': { 
 

 
      frac_role: user.frac_role, 
      division: user.division, 
      state: user.state 
    }, 
    'https://purl.imsglobal.org/spec/lti-ags/claim/endpoint': { 
      scope: ['https://purl.imsglobal.org/spec/lti-ags/scope/score'], 
      lineitem: `http://localhost:${PORT}/platform/ags/lineitem/competency_assessment` 
    } 
  }; 
 
  const idToken = jwt.sign(payload, privateKey, { algorithm: 'RS256', keyid: 'mock-platform-key-1' 
}); 
 
  // Auto-submitting HTML form simulating an LMS POST redirect to KarmMitra backend 
  res.send(` 
    <html> 
      <body onload="document.forms[0].submit()"> 
        <form method="POST" action="http://localhost:8000/lti/launch"> 
          <input type="hidden" name="id_token" value="${idToken}" /> 
        </form> 
        <p>Connecting to KarmMitra AI Sovereign Engine via LTI 1.3 Handshake...</p> 
      </body> 
    </html> 
  `); 
}); 
 
// 4. LTI 1.3 Assignment and Grade Services (AGS) Receiver 
app.post('/platform/ags/lineitem/competency_assessment/scores', (req, res) => { 
  const gradePayload = req.body; 
  console.log('\n[MOCK iGOT PLATFORM] Successfully received AGS Grade Passback:'); 
  console.log(JSON.stringify(gradePayload, null, 2)); 
 
  res.status(200).json({ 
    status: 'success', 
    message: 'Grade recorded securely into simulated iGOT gradebook', 
    timestamp: new Date().toISOString() 
  }); 
}); 
 
app.listen(PORT, () => { 
  console.log(`Mock iGOT Karmayogi Platform listening on http://localhost:${PORT}`); 
});
